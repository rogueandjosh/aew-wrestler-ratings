function backfillAnalysis() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const analysisSheet = ss.getSheetByName('Weekly Analysis'); const props = PropertiesService.getScriptProperties(); const token = props.getProperty('GITHUB_TOKEN'); const owner = 'rogueandjosh'; const repo = 'aew-wrestler-ratings'; const data = analysisSheet.getDataRange().getValues(); const labels = ['overall', 'mens', 'womens', 'menstag', 'womenstag', 'trios']; // Process every row that has any content in columns B-G for (let i = 1; i < data.length; i++) { const hasContent = data[i].slice(1, 7).some(cell => cell && cell.toString().trim() !== ''); if (!hasContent) continue; const weekLabel = data[i][0]; if (!weekLabel) continue; const sections = {}; data[i].slice(1, 7).forEach((cell, j) => { const text = cell ? cell.toString().trim() : ''; if (text !== '') sections[labels[j]] = text; }); const jsonData = { week: weekLabel, generated: new Date().toISOString(), sections: sections }; const content = JSON.stringify(jsonData, null, 2); const archiveFilename = weekLabelToArchiveSlug(weekLabel); const r1 = uploadToGitHub(token, owner, repo, 'analysis.json', content, `Update analysis for ${weekLabel}`); Logger.log(r1.success ? `✅ analysis.json uploaded for ${weekLabel}` : `❌ Failed: ${r1.error}`); if (archiveFilename) { const r2 = uploadToGitHub(token, owner, repo, `analysis-archive/${archiveFilename}`, content, `Archive analysis for ${weekLabel}`); Logger.log(r2.success ? `✅ Archived ${archiveFilename}` : `❌ Archive failed: ${r2.error}`); } Utilities.sleep(1000); // Avoid rate limiting between uploads } Logger.log('✅ Backfill complete'); }
